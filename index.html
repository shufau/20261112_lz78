[files] "lz78.py" = "./lz78.py" import js from pyscript import window from pyodide.ffi import to_js # 從 lz78.py 引入壓縮與解壓縮函式 from lz78 import compress_data, decompress_data def js_compress_data(js_uint8_array): input_data = bytes(js_uint8_array.to_bytes()) res = compress_data(input_data) return to_js(res) def js_decompress_data(js_uint8_array): input_data = bytes(js_uint8_array.to_bytes()) res = decompress_data(input_data) return to_js(res) window.py_compress = js_compress_data window.py_decompress = js_decompress_data # 通知 JS 前端 Python 模組已經準備完畢 window.onPythonReady()
