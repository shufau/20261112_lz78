import js
from pyscript import window
from pyodide.ffi import to_js

# 匯入兩組演算法
import lz78
import adaptive_huffman

# LZ78 綁定
def js_lz78_compress(js_uint8_array):
    input_data = bytes(js_uint8_array.to_bytes())
    res = lz78.compress_data(input_data)
    return to_js(res)

def js_lz78_decompress(js_uint8_array):
    input_data = bytes(js_uint8_array.to_bytes())
    res = lz78.decompress_data(input_data)
    return to_js(res)

# Adaptive Huffman 綁定
def js_ah_compress(js_uint8_array):
    input_data = bytes(js_uint8_array.to_bytes())
    res = adaptive_huffman.compress_data(input_data)
    return to_js(res)

def js_ah_decompress(js_uint8_array):
    input_data = bytes(js_uint8_array.to_bytes())
    res = adaptive_huffman.decompress_data(input_data)
    return to_js(res)

# 註冊至全域 window 物件
window.py_lz78_compress = js_lz78_compress
window.py_lz78_decompress = js_lz78_decompress
window.py_ah_compress = js_ah_compress
window.py_ah_decompress = js_ah_decompress

# 通知 JS 前端 Python 模組準備完畢
window.onPythonReady()
